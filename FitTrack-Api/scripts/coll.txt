NAME: Register User
  POST {{baseUrl}}/api/v1/auth/register
  BODY: { "firstName": "Jane", "lastName": "Smith", "email": "jane.smith@fittrack.com", "username": "janesmith", "password": "secret123" }
  EVENT(test):
    pm.test('Status is 201', () => pm.response.to.have.status(201));
    pm.test('Success is true', () => pm.expect(pm.response.json().success).to.eql(true));
    pm.test('Token is generated', () => {
        const token = pm.response.json().data.token;
        pm.expect(token).to.be.a('string').and.not.empty;
        pm.collectionVariables.set('token', token);
    });
    pm.test('User object returned without password', () => {
        const user = pm.response.json().data.user;
        pm.expect(user).to.have.property('email');
        pm.expect(user).to.not.have.property('password');
    });
